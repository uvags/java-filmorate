  "id" BIGINT [pk, increment]

  "name" VARCHAR(255)
  "birthday" DATE

  "id" BIGINT [pk, increment]
  "name" VARCHAR(255) [not null]
  "description" VARCHAR(200)
  "release_date" DATE [not null]
  "duration" INT [not null]
  "rating_id" BIGINT
}

Table "likes" {

  Indexes {

  }
}

Table "friendships" {
  "user_id" BIGINT [not null]
  "friend_id" BIGINT [not null]
  "status" VARCHAR(20) [not null]

  Indexes {
